app.post('/processar-pagamento', async (req, res) => { try { const { transaction_amount, nomeCliente, cpfCliente, emailCliente, tipoIngresso } = req.body; const result = await payment.create({ body: { transaction_amount: Number(transaction_amount), description: `Vibe Privê - ${tipoIngresso}`, payment_method_id: 'pix', payer: { email: emailCliente, first_name: nomeCliente, identification: { type: 'CPF', number: cpfCliente } }, metadata: { nome_cliente: nomeCliente, email_cliente: emailCliente, tipo_ingresso: tipoIngresso }, notification_url: 'https://onrender.com' } }); res.json({ id: result.id, status: result.status, qr_code: result.point_of_interaction?.transaction_data?.qr_code, qr_code_base64: result.point_of_interaction?.transaction_data?.qr_code_base64 }); } catch (error) { console.error(error); res.status(500).json({ error: 'Erro ao gerar Pix' }); } });
